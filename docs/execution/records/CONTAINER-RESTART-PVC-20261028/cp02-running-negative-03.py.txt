 assert not fields.keys()-names.keys()

 sql="""BEGIN READ ONLY;

 'pending',snapshot->'Pending','fence',recovery_fence,
 'operation_count',(SELECT count(*) FROM instance.instance_operations o WHERE o.tenant_id=i.tenant_id AND o.instance_id=i.instance_id),

 FROM instance.instances i WHERE tenant_id='64c5c35a-f554-432c-b4bc-7b4f36f08420' AND instance_id='e3dd2bc7-1c6a-429b-a401-e175c287d5cf';
COMMIT;"""
 q=subprocess.run([str(r/'pg18-client-01/bin/psql'),'-X','-w','-qAt','-v','ON_ERROR_STOP=1','-v','VERBOSITY=terse'],input=sql,text=True,env=env,capture_output=True,timeout=25)

  e.state['after_'+label]=current();observed=pg();d.require(observed==e.state['pg_before'],'REJECTED_INTENT_CHANGED_DURABLE_STATE')

 e.state['result']='PASS';e.save()
finally:
