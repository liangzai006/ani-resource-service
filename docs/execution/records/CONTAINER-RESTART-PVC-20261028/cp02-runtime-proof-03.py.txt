PRIVATE_STDERR = None

def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':')).encode()).hexdigest()

def control(value):
    value=copy.deepcopy(value);value.pop('status',None)
    for key in ('resourceVersion','managedFields'):value.get('metadata',{}).pop(key,None)
    return value

def k(argv, absent=False):
    cmd = ['kubectl', '--context', CTX, '--request-timeout=12s'] + argv
    p = subprocess.run(['ssh', '-oBatchMode=yes', '-oStrictHostKeyChecking=yes', '-oConnectTimeout=10',
        'kubercloud@10.10.1.66', shlex.join(cmd)], capture_output=True, timeout=30, check=False)
    if p.returncode:
        if PRIVATE_STDERR is not None:
            fd=os.open(PRIVATE_STDERR,os.O_CREAT|os.O_EXCL|os.O_WRONLY,0o600)
            with os.fdopen(fd,'wb') as f:f.write(p.stderr[:1048576])
        raise subprocess.CalledProcessError(p.returncode,cmd)
    return None if absent and not p.stdout.strip() else json.loads(p.stdout)

def get(kind, name, namespaced=True, absent=False):
    return k(['get', kind, name] + (['-n', NS] if namespaced else []) +
        (['--ignore-not-found'] if absent else []) + ['-o', 'json'], absent)

def identity(x, uid):
    assert x['metadata']['uid'] == uid and not x['metadata'].get('deletionTimestamp')
    return {'uid': uid, 'generation': x['metadata'].get('generation')}

def owner(x, uid):
    kind = 'Notebook' if uid == NB_UID else 'StatefulSet'
    assert any(v.get('controller') and v.get('uid') == uid and v.get('kind') == kind and v.get('name') == NAME for v in x['metadata'].get('ownerReferences', []))

def template(spec, attached):
    volumes = {v['name']: v['persistentVolumeClaim'] for v in spec.get('volumes', []) if 'persistentVolumeClaim' in v}
    assert sorted(v['claimName'] for v in volumes.values()) == sorted([WORK] + ([BORROW] if attached else []))
    containers = [c for c in spec['containers'] if c['name'] == 'workspace']; assert len(containers) == 1
    mounts = [m for m in containers[0].get('volumeMounts', []) if m['name'] in volumes]
    assert len({m['mountPath'] for m in mounts}) == len(mounts)
    result = {m['mountPath']: [volumes[m['name']]['claimName'], bool(m.get('readOnly')), bool(volumes[m['name']].get('readOnly'))] for m in mounts}
    assert all(not m.get('subPath') and not m.get('subPathExpr') for m in mounts)
    assert result == dict({'/workspace': [WORK, False, False]}, **({TARGET: [BORROW, False, False]} if attached else {}))
    return digest(result)

def storage():
    result = {}
    for name, uid in ((WORK, WORK_UID), (BORROW, BORROW_UID)):
        pvc = get('pvc', name); identity(pvc, uid); assert pvc['status']['phase'] == 'Bound'
        pv_name = pvc['spec']['volumeName']; pv = get('pv', pv_name, False); identity(pv, pv['metadata']['uid'])
        assert pv['status']['phase'] == 'Bound' and pv['spec']['claimRef']['uid'] == uid
        assert pv['spec']['claimRef']['namespace'] == NS and pv['spec']['claimRef']['name'] == name
        if name == BORROW: assert pv_name == 'pvc-fa56c965-5a9a-4bb1-b271-3960a7309e96'
        result[name] = {'pvc_uid': uid, 'pv_name': pv_name, 'pv_uid': pv['metadata']['uid'], 'phase': 'Bound'}
    return result

def main():
    global PRIVATE_STDERR
    p = argparse.ArgumentParser(description=__doc__); p.add_argument('--execute', action='store_true')
    p.add_argument('--mode', choices=MODES); p.add_argument('--out'); p.add_argument('--baseline'); p.add_argument('--previous-pod-uid')
    a = p.parse_args()
    if not a.execute: print(json.dumps({'result': 'PREPARED_ONLY', 'modes': MODES, 'remote_calls': 0})); return 0
    if not a.mode or not a.out: p.error('--execute requires --mode and --out')
    assert os.geteuid() == 0 and pathlib.Path(__file__).resolve().is_relative_to(ROOT)
    out = pathlib.Path(a.out).resolve(); assert out.is_relative_to(ROOT) and out.name.endswith('.safe.json')
    PRIVATE_STDERR=out.with_name(out.name+'.private.stderr')
    if a.previous_pod_uid: assert str(uuid.UUID(a.previous_pod_uid)) == a.previous_pod_uid
    fd = os.open(out, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    running, attached = a.mode.startswith('running-'), a.mode.endswith('-attached')
    rec = {'schema': 'cp02-runtime-proof-03', 'result': 'FAIL', 'mode': a.mode, 'observed_at': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'marker_writes': 0}
    try:
        rec['stage'] = 'cluster'; cfg = k(['config', 'view', '--minify', '-o', 'json'])

        cluster = cfg['clusters'][0]['cluster']; assert cluster['server'].startswith('https://') and not cluster.get('insecure-skip-tls-verify')
        identity(get('namespace', 'kube-system', False), SYSTEM_UID); identity(get('namespace', NS, False), NS_UID)
        rec['cluster'] = {'context': CTX, 'server': cluster['server'], 'kube_system_uid': SYSTEM_UID, 'namespace': NS, 'namespace_uid': NS_UID}
        rec['stage'] = 'storage'; rec['storage'] = storage()
        if a.baseline:
            raw = pathlib.Path(a.baseline).read_bytes(); base = json.loads(raw)
            assert base['schema'] == rec['schema'] and base['result'] == 'PASS' and base['cluster'] == rec['cluster'] and base['storage'] == rec['storage']
            rec['baseline_sha256'] = hashlib.sha256(raw).hexdigest()
        rec['stage'] = 'templates'; nb, sts = get('notebook', NAME), get('statefulset', NAME)
        rec['notebook'] = identity(nb, NB_UID); rec['statefulset'] = identity(sts, STS_UID); owner(sts, NB_UID)
        assert nb['metadata'].get('annotations', {}).get('kubeflow-resource-stopped') in ((None, 'false') if running else ('true',))
        assert sts['spec']['replicas'] == int(running) and sts['status']['observedGeneration'] == sts['metadata']['generation']
        if running: assert sts['status'].get('readyReplicas', 0) == 1
        rec['template_sha256'] = {'notebook': template(nb['spec']['template']['spec'], attached), 'statefulset': template(sts['spec']['template']['spec'], attached)}
        listing = k(['get', 'pods', '-n', NS, '-o', 'json']); assert not listing.get('metadata', {}).get('continue')
        own = [x for x in listing['items'] if x['metadata']['name'].startswith(NAME + '-') or any(o.get('uid') in (NB_UID, STS_UID) for o in x['metadata'].get('ownerReferences', [])) or x['metadata'].get('labels', {}).get('notebook-name') == NAME]
        pod = get('pod', NAME + '-0', absent=True); rec['stage'] = 'runtime' if running else 'stopped'
        if not running:
            assert not own and pod is None
            rec['pod_absence'] = {'complete_namespace_list': True, 'target_count': 0, 'direct_name': NAME + '-0', 'direct_not_found': True, 'previous_pod_uid': a.previous_pod_uid}
        else:
            assert len(own) == 1 and pod and own[0]['metadata']['uid'] == pod['metadata']['uid']; owner(pod, STS_UID)
            uid = pod['metadata']['uid']; rec['pod'] = identity(pod, uid); assert pod['status']['phase'] == 'Running'
            assert not a.previous_pod_uid or uid != a.previous_pod_uid
            assert any(x['type'] == 'Ready' and x['status'] == 'True' for x in pod['status'].get('conditions', []))
            assert any(x['name'] == 'workspace' and x['ready'] for x in pod['status'].get('containerStatuses', []))
            rec['template_sha256']['pod'] = template(pod['spec'], attached)
