package sandbox
import("context";"testing")
func TestCrossReviewSuccessfulOldDeleteCompletesResumeBeforeReady(t *testing.T){
 w,r,p:=testWorker(t,DesiredRunning)
 r.work.Sandbox.Version=2
 r.work.Sandbox.LastOperationID="22222222-2222-4222-8222-222222222222"
 pending:=Mutation{ID:"33333333-3333-4333-8333-333333333333",Kind:MutationDeletePod,Version:1,OperationID:"11111111-1111-4111-8111-111111111111",TargetUID:"old-pod"}
 r.work.Sandbox.Pending=&pending
 p.observe=Observation{Observed:ObservedRunning,WorkspaceExists:true,WorkspaceBound:true,PodExists:true,ProviderRefs:ProviderRefs{PodUID:"old-pod",PVCUID:"original-workspace"}}
 providerPodPresent:=true
 p.apply=func(_ context.Context,_ Sandbox,m Mutation)(MutationResult,error){providerPodPresent=false;return MutationResult{Mutation:m,Outcome:MutationApplied},nil}
 step(t,w)
 if len(p.calls)!=1||p.calls[0].Kind!=MutationDeletePod||providerPodPresent||r.work.Sandbox.Pending!=nil||r.progress.Sandbox.Observed!=ObservedRunning{t.Fatalf("actual behavior changed: calls=%v present=%v progress=%#v",p.calls,providerPodPresent,r.progress)}
 t.Logf("verified defect: successful retry deletes old Pod (present=%v), but current Resume v%d progress remains desired=%s observed=%s pending=%v; Postgres.Finish's succeeded condition is already true before replacement exists/Ready",providerPodPresent,r.progress.Sandbox.Version,r.progress.Sandbox.Desired,r.progress.Sandbox.Observed,r.work.Sandbox.Pending)
}
